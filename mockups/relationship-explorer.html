Cost relationship explorer

Three views of the links already in the warehouse: how cost rolls up the WBS, which vendors and purchase orders sit behind it, and where a link is missing. Every figure below would be the output of a stored query; the SQL for each view is under it.

Blue bars are under budget, red bars are over. An element with no budget is shown as “No budget”, never as an overrun the size of its spend. Amounts in thousands.

SQL · wbs_rollup
WITH RECURSIVE tree(wbs_id, root_id, depth) AS (
  SELECT wbs_id, wbs_id, 0 FROM dim_wbs
  UNION ALL
  SELECT c.wbs_id, t.root_id, t.depth + 1
  FROM dim_wbs c JOIN tree t ON c.parent_wbs_id = t.wbs_id
)
SELECT w.wbs_code, w.description,
       SUM(b.amount) AS budget,           -- NULL when no budget row exists
       SUM(a.amount) AS actual,
       SUM(a.amount) - SUM(b.amount) AS variance
FROM tree
JOIN dim_wbs w ON w.wbs_id = tree.root_id
LEFT JOIN v_actual a ON a.wbs_id = tree.wbs_id   -- cost only, POSTED batches
LEFT JOIN v_budget b ON b.wbs_id = tree.wbs_id
GROUP BY w.wbs_id;

SQL · vendor_po_wbs_edges
-- vendor → PO : the detail behind the PO (never added to actuals)
SELECT vendor AS source, po_no AS target, SUM(amount) AS weight
FROM v_service_line GROUP BY vendor, po_no;

-- PO → WBS : where the PO's money was actually posted
SELECT po_no AS source, wbs_code AS target, SUM(amount) AS weight
FROM v_actual WHERE po_no IS NOT NULL GROUP BY po_no, wbs_code;

SQL · po_reconciliation_gaps
WITH act AS (SELECT po_no, SUM(amount) AS actual FROM v_actual WHERE po_no IS NOT NULL GROUP BY po_no),
     det AS (SELECT po_no, SUM(amount) AS detail FROM v_service_line GROUP BY po_no)
SELECT COALESCE(act.po_no, det.po_no) AS po_no, act.actual, det.detail,
       CASE WHEN det.po_no IS NULL THEN 'Posted, no detail'
            WHEN act.po_no IS NULL THEN 'Detail, nothing posted'
            WHEN ABS(act.actual - det.detail) > 1 THEN 'Amounts differ' END AS gap
FROM act LEFT JOIN det USING (po_no)
UNION
SELECT det.po_no, NULL, det.detail, 'Detail, nothing posted'
FROM det LEFT JOIN act USING (po_no) WHERE act.po_no IS NULL;
